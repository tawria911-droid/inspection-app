async function sendPDFViaEmail(e) { if(e) e.preventDefault(); // 1. جلب البيانات الأساسية const facName = document.getElementById('facilityName')?.value.trim() || ''; const inspector = document.getElementById('inspectorName')?.value.trim() || ''; if(!facName || !inspector) { alert("⚠️ يرجى تعبئة اسم المسجد/المنشأة واسم معد التقرير أولاً."); return; } const btn = document.getElementById('sendEmailBtn'); const originalHTML = btn.innerHTML; btn.disabled = true; btn.innerHTML = ` جاري إرسال التقرير...`; try { // 2. تجميع بيانات كافة البنود والملاحظات تلقائياً let reportDetails = ""; const items = document.querySelectorAll('.item-card, tr[data-item], .check-item'); if (items.length > 0) { items.forEach((item, index) => { const title = item.querySelector('.item-title, h4, label, td:first-child')?.innerText.trim() || `البند ${index + 1}`; const statusSelect = item.querySelector('select'); const statusRadio = item.querySelector('input[type="radio"]:checked'); const status = statusSelect ? statusSelect.value : (statusRadio ? statusRadio.value : "مكتمل"); const noteInput = item.querySelector('textarea, input[type="text"]:not(#facilityName):not(#inspectorName)'); const note = noteInput && noteInput.value.trim() ? noteInput.value.trim() : "لا توجد ملاحظات"; reportDetails += `${index + 1}. ${title}\n • الحالة: ${status}\n • الملاحظات: ${note}\n-----------------------------------\n`; }); } else { reportDetails = "تم إرسال التقرير بنجاح عبر النظام."; } // 3. تجهيز البيانات للارسال const templateParams = { facility_name: facName, inspector_name: inspector, inspection_date: document.getElementById('inspectionDate')?.value || "غير محدد", location: document.getElementById('location')?.value || "غير محدد", gps_coords: document.getElementById('gpsCoords')?.value || "غير محدد", serial_no: document.getElementById('serialNo')?.value || "غير محدد", contractor: document.getElementById('maintenanceContractor')?.value || "غير محدد", report_details: reportDetails }; // 4. الإرسال عبر EmailJS await emailjs.send(EMAILJS_SERVICE_ID, EMAILJS_TEMPLATE_ID, templateParams); alert("✅ تم إرسال التقرير بنجاح إلى إيميل الوحدة!"); } catch (error) { console.error("خطأ في الإرسال:", error); alert("❌ حدث خطأ أثناء الإرسال: " + (error.text || JSON.stringify(error))); } finally { btn.disabled = false; btn.innerHTML = originalHTML; } }
