async function sendPDFViaEmail(e) { if (e) e.preventDefault(); // 1. جلب البيانات الأساسية const facName = document.getElementById('facilityName')?.value.trim() || ''; const inspector = document.getElementById('inspectorName')?.value.trim() || ''; if (!facName || !inspector) { alert("⚠️ يرجى تعبئة اسم المسجد/المنشأة واسم معد التقرير أولاً."); return; } const btn = document.getElementById('sendEmailBtn') || document.querySelector('button[type="submit"]'); const originalHTML = btn ? btn.innerHTML : ''; if (btn) { btn.disabled = true; btn.innerHTML = ` جاري إرسال التقرير...`; } try { // 2. تجميع بيانات بنود الفحص والملاحظات تلقائياً من أي تصميم let reportDetails = ""; // البحث عن عناصر البنود في التصميم const itemRows = document.querySelectorAll('.item-card, tr[data-item], .check-item, .form-group, fieldset, .card'); itemRows.forEach((item, index) => { const titleEl = item.querySelector('.item-title, h4, label, td:first-child, strong'); const statusEl = item.querySelector('select, input[type="radio"]:checked'); const noteEl = item.querySelector('textarea, input[type="text"]:not(#facilityName):not(#inspectorName):not(#location):not(#gpsCoords):not(#serialNo):not(#maintenanceContractor)'); if (titleEl && (statusEl || noteEl)) { const title = titleEl.innerText.trim(); const status = statusEl ? (statusEl.value || statusEl.nextSibling?.textContent?.trim()) : "سليم"; const note = noteEl && noteEl.value.trim() ? noteEl.value.trim() : "لا توجد ملاحظات"; reportDetails += `${title}\n • الحالة: ${status}\n • الملاحظات: ${note}\n-----------------------------------\n`; } }); if (!reportDetails) { reportDetails = "تم إرسال التقرير الميداني بنجاح."; } // 3. إعداد البيانات الممررة لـ EmailJS const templateParams = { facility_name: facName, inspector_name: inspector, inspection_date: document.getElementById('inspectionDate')?.value || "غير محدد", location: document.getElementById('location')?.value || "غير محدد", gps_coords: document.getElementById('gpsCoords')?.value || "غير محدد", serial_no: document.getElementById('serialNo')?.value || "غير محدد", contractor: document.getElementById('maintenanceContractor')?.value || "غير محدد", report_details: reportDetails }; // 4. الإرسال await emailjs.send("service_w1kg7u8", "template_k0n0gd8", templateParams); alert("✅ تم إرسال التقرير بنجاح إلى إيميل الوحدة!"); } catch (error) { console.error("خطأ في الإرسال:", error); alert("❌ حدث خطأ أثناء الإرسال: " + (error.text || JSON.stringify(error))); } finally { if (btn) { btn.disabled = false; btn.innerHTML = originalHTML; } } }
